# 프론트엔드 데이터 모델 (TypeScript)

백엔드 DTO를 그대로 옮긴 타입 정의. `src/types/api.ts` 등에 복사해 사용한다.

> 주석의 ⚠️ 는 실제 값이 문서와 다르게 동작하는 지점이다.

```ts
// ─────────────────────────────────────────────
// 공통
// ─────────────────────────────────────────────

/** 모든 실패 응답의 공통 바디 */
export interface ErrorResponse {
  code: ErrorCode;
  message: string;
  timestamp: string; // "2026-08-01T14:32:11.123"
}

/** Spring Data Page (VIA_DTO 직렬화 모드) */
export interface Page<T> {
  content: T[];
  page: {
    size: number;
    number: number;         // 0-based
    totalElements: number;
    totalPages: number;
  };
}

export const isLastPage  = (p: Page<unknown>) => p.page.number + 1 >= p.page.totalPages;
export const isFirstPage = (p: Page<unknown>) => p.page.number === 0;

/** 회원가입 등 범용 결과 */
export interface IngestResult {
  status: string;   // "ok"
  message: string | null;
}

// ─────────────────────────────────────────────
// 인증
// ─────────────────────────────────────────────

export interface SignupRequest {
  email: string;
  password: string;      // 8~30자
  nick_name: string;     // ⚠️ 스네이크 케이스
  role: 'USER';          // ⚠️ 클라이언트가 지정. 항상 'USER' 고정 전송
}

export interface LoginRequest {
  email: string;
  password: string;      // 8자 이상
}

/** 로그인 / 소셜 로그인 응답 */
export interface UserResponse {
  id: number;
  email: string;
  nickName: string;
  accessToken: string;   // ⚠️ "Bearer " 접두사 포함됨
  refreshToken: string;  // ⚠️ 디버그용. 쿠키를 신뢰할 것
  role: string | null;   // ⚠️ 항상 null (미설정)
}

/** POST /api/auth/reissue 응답 */
export interface TokenResponse {
  accessToken: string;   // ⚠️ "Bearer " 접두사 없음!
  refreshToken: string;  // 기존 값 그대로 (회전 없음)
}

// ─────────────────────────────────────────────
// 사용자
// ─────────────────────────────────────────────

/** GET /api/user/me — ⚠️ 현재 서버에서 동작하지 않음 */
export interface UserProfileResponse {
  nickName: string;
  phoneNumber: string | null;
  birth: string;         // ⚠️ null이면 빈 문자열 ""
  createdAt: string;
}

// ─────────────────────────────────────────────
// 게시판
// ─────────────────────────────────────────────

export interface BoardRequest {
  name: string;
  description?: string;
}

export interface BoardResponse {
  id: number;
  name: string;
  description: string | null;
  createdAt: string;
}

// ─────────────────────────────────────────────
// 게시글
// ─────────────────────────────────────────────

export interface PostRequest {
  title: string;   // 5~200자
  body: string;
}

export interface PostImageResponse {
  id: number;
  url: string;            // "/images/{storedName}" — API_BASE를 앞에 붙여 사용
  originalName: string;
  sortOrder: number;
}

export interface PostDTO {
  id: number;
  title: string;
  author: string;         // ⚠️ 닉네임 문자열. userId 아님
  board: string;          // ⚠️ 게시판 이름 문자열. boardId 아님
  body: string;
  viewCount: number;
  images: PostImageResponse[];
  like: number;           // 좋아요 개수 (상세·목록 모두 채워짐)
  disLike: number;        // 싫어요 개수
  myReaction: ReactionType | null;  // 비로그인이거나 반응 없으면 null
  createdAt: string;
  canEdit: boolean;       // ⚠️ 목록/작성 응답에서는 항상 false
  canDelete: boolean;
}

// ─────────────────────────────────────────────
// 댓글
// ─────────────────────────────────────────────

export interface CommentCreateRequest {
  content: string;        // 최대 1000자
  parentId?: number;      // 없으면 최상위 댓글
}

export interface CommentResponse {
  id: number;
  authorUsername: string; // ⚠️ 닉네임만. 작성자 ID 없음 → 내 댓글 판별 불가
  content: string;        // deleted면 "삭제된 게시글입니다"
  deleted: boolean;
  createdAt: string;
  likeCount: number;
  dislikeCount: number;
  myReaction: ReactionType | null;  // 비로그인이거나 반응 없으면 null
  children: CommentResponse[];  // 최상위 댓글에만 채워짐
  // ⚠️ parentId 필드 없음
}

export const DELETED_COMMENT_CONTENT = '삭제된 게시글입니다';

// ─────────────────────────────────────────────
// 반응
// ─────────────────────────────────────────────

export type ReactionType = 'LIKE' | 'DISLIKE';

export interface ReactionRequest {
  type: ReactionType;
}

export interface ReactionResponse {
  likeCount: number;
  dislikeCount: number;
  myReaction: ReactionType | null;  // 취소 직후 null
}

// ─────────────────────────────────────────────
// 알림
// ─────────────────────────────────────────────

export type NotificationType = 'COMMENT_ON_POST' | 'REPLY_ON_COMMENT';

export interface NotificationResponse {
  id: number;
  type: NotificationType;
  message: string;        // 서버가 조립한 한국어 문장
  actorUsername: string;
  postId: number;         // ⚠️ FK 아님 — 삭제된 게시글일 수 있음
  commentId: number | null;
  read: boolean;
  createdAt: string;
}

export interface UnreadCountResponse {
  count: number;
}

// ─────────────────────────────────────────────
// 에러 코드
// ─────────────────────────────────────────────

export type ErrorCode =
  // 404
  | 'USER_NOT_FOUND' | 'BOARD_NOT_FOUND' | 'POST_NOT_FOUND' | 'COMMENT_NOT_FOUND'
  // 409
  | 'DUPLICATE_USER_EMAIL' | 'DUPLICATE_BOARD_NAME'
  // 403
  | 'ACCESS_DENIED' | 'POST_ACCESS_DENIED' | 'BOARD_ACCESS_DENIED' | 'CANNOT_VIEW_NOTIFICATION'
  // 401
  | 'LOGIN_REQUIRED' | 'LOGIN_FAILED' | 'INVALID_OAUTH_STATE'
  // 400
  | 'INVALID_INPUT' | 'INVALID_BOARD_ID'
  | 'INVALID_FILE_TYPE' | 'FILE_COUNT_EXCEEDED' | 'MAX_UPLOAD_SIZE_EXCEEDED'
  | 'CANNOT_REPLY_TO_REPLY' | 'CANNOT_REPLY_TO_DELETED' | 'CANNOT_EDIT_DELETED'
  | 'COMMENT_POST_MISMATCH'
  | 'SQL_INTEGRITY_ERROR' | 'METHOD_NOT_ALLOWED'
  // 500
  | 'INVALID_FILE_UPLOAD_DIR' | 'FILE_UPLOAD_FAILED' | 'INTERNAL_SERVER_ERROR';
```

## 검증 스키마 (Zod)

서버가 필드 단위 에러를 주지 않으므로 클라이언트 검증이 필수다.
→ [[API-에러코드]]

```ts
import { z } from 'zod';

export const signupSchema = z.object({
  email:    z.string().email('올바른 이메일 형식이 아닙니다'),
  password: z.string().min(8, '8자 이상').max(30, '30자 이하'),
  passwordConfirm: z.string(),
  nickName: z.string().min(1, '닉네임을 입력하세요'),
}).refine(v => v.password === v.passwordConfirm, {
  message: '비밀번호가 일치하지 않습니다',
  path: ['passwordConfirm'],
});

export const loginSchema = z.object({
  email:    z.string().min(1, '이메일을 입력하세요'),
  password: z.string().min(8, '8자 이상'),
});

export const postSchema = z.object({
  title: z.string().min(5, '제목은 5자 이상').max(200, '제목은 200자 이하'),
  body:  z.string().min(1, '본문을 입력하세요'),
});

export const boardSchema = z.object({
  name: z.string().min(1, '게시판 이름을 입력하세요'),
  description: z.string().optional(),
});

export const commentSchema = z.object({
  content: z.string().min(1, '내용을 입력하세요').max(1000, '1000자 이하'),
});

// 이미지
export const ALLOWED_IMAGE_TYPES = ['image/png', 'image/jpeg', 'image/gif', 'image/webp'] as const;
export const MAX_IMAGE_SIZE  = 2 * 1024 * 1024;   // 2MB
export const MAX_IMAGE_COUNT = 5;
```

## 요청 → 응답 매핑

| 요청 타입 | 엔드포인트 | 응답 타입 |
|---|---|---|
| `SignupRequest` | `POST /api/auth/signup` | `IngestResult` |
| `LoginRequest` | `POST /api/auth/login` | `UserResponse` |
| — | `POST /api/auth/reissue` | `TokenResponse` |
| — | `POST /api/auth/logout` | (없음) |
| — | `GET /api/user/me` | `UserProfileResponse` |
| — | `GET /api/board/all` | `BoardResponse[]` |
| `BoardRequest` | `POST /api/board/new` | `BoardResponse` |
| `BoardRequest` | `PUT /api/board/{id}/update` | `BoardResponse` |
| — | `DELETE /api/board/{id}` | `string` ("ok", **text/plain**) |
| — | `GET /api/post/all` | `PostDTO[]` |
| — | `GET /api/post/{boardId}/all` | `Page<PostDTO>` |
| — | `GET /api/post/{id}` | `PostDTO` |
| multipart | `POST /api/post/{boardId}/new` | `PostDTO` |
| `PostRequest` | `PUT /api/post/{id}/update` | `PostDTO` |
| — | `DELETE /api/post/{id}` | `string` ("ok", **text/plain**) |
| — | `GET /api/comment/post/{postId}/list` | `Page<CommentResponse>` |
| `CommentCreateRequest` | `POST /api/comment/post/{postId}/new` | `CommentResponse` |
| `CommentCreateRequest` | `PUT /api/comment/{id}` | `CommentResponse` |
| — | `DELETE /api/comment/{id}` | (없음) |
| `ReactionRequest` | `POST /api/post/{postId}/reation` | `ReactionResponse` |
| `ReactionRequest` | `POST /api/comment/{commentId}/reaction` | `ReactionResponse` |
| — | `GET /api/notify/list` | `Page<NotificationResponse>` |
| — | `GET /api/notify/unreads` | `UnreadCountResponse` |
| — | `PUT /api/notify/{id}/read` | (없음) |
